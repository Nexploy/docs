---
title: Useful commands
description: Common development commands
---

```bash
# Development
pnpm dev                                # All apps
pnpm dev:nexploy                        # Next.js only (port 3000)
pnpm dev:docker-api                     # Docker API only (port 3300)

# Build
pnpm build                              # All apps

# Database
pnpm --filter=nexploy db:migrate:dev    # Create & apply migrations
pnpm --filter=nexploy db:migrate:prod   # Apply migrations (prod)
pnpm --filter=nexploy db:generate       # Generate the Prisma client
pnpm --filter=nexploy db:studio         # Prisma Studio (GUI)
pnpm --filter=nexploy db:reset          # Reset the database

# Code quality
pnpm lint                               # Lint
pnpm types                              # Type check
pnpm format                             # Prettier

# e2e tests (Playwright)
pnpm test:e2e                           # Run the tests
pnpm test:e2e:ui                        # UI mode
```
